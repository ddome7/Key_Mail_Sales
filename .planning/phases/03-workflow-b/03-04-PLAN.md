    ```typescript
    /**
     * Gmail 스레드 전문 로더.
     *
     * 실제 Gmail MCP 도구 이름(gmail_read_thread / gmail_get_thread / ...)은
     * Plan 05 오케스트레이터에서 확정한다. 이 모듈은 DI 인터페이스만 정의.
     */

    export interface ThreadMessage {
      messageId: string;
      from: string;
      to: string;
      subject: string;
      body: string;          // plain text 또는 HTML (Gmail raw)
      internalDate: string;  // Gmail internalDate (ms epoch as string)
    }

    /**
     * Gmail MCP의 스레드 조회 도구를 감싼 DI 함수 타입.
     * Plan 05에서 실제 MCP 호출 함수를 이 인터페이스로 주입.
     *
     * raw 응답에서 ThreadMessage로 정규화하는 책임은 주입 측에 있지만,
     * 방어적으로 loadThreadHistory 내부에서도 필수 필드 기본값을 채운다.
     */
    export type GmailReadThreadFn = (args: {
      threadId: string;
    }) => Promise<ThreadMessage[]>;

    /**
     * 기존 스레드 이력 로드.
     *
     * @param threadId - Gmail threadId. undefined/null/'' → 빈 배열 (새 이메일)
     * @param gmailReadThreadTool - DI Gmail MCP 도구
     * @returns 시간순(오래된 → 최신) 정렬된 메시지 배열
     *
     * **D-11 처리:** threadId가 없으면 "신규 스레드"로 간주하여 빈 배열 반환.
     * Gmail API가 빈 문자열 threadId를 거부하므로(Pitfall 간접), 아예 호출 생략.
     */
    export async function loadThreadHistory(
      threadId: string | null | undefined,
      gmailReadThreadTool: GmailReadThreadFn,
    ): Promise<ThreadMessage[]> {
      if (!threadId || threadId.trim() === '') return [];

      const raw = await gmailReadThreadTool({ threadId });

      // 방어적 정규화 + 시간순 정렬
      const normalized = raw.map(m => ({
        messageId: m.messageId ?? '',
        from: m.from ?? '',
        to: m.to ?? '',
        subject: m.subject ?? '',
        body: m.body ?? '',
        internalDate: m.internalDate ?? '0',
      }));

      // internalDate는 ms epoch 문자열 → 숫자 비교
      normalized.sort((a, b) => {
        const na = parseInt(a.internalDate, 10) || 0;
        const nb = parseInt(b.internalDate, 10) || 0;
        return na - nb;
      });

      return normalized;
    }
    ```

    ```typescript
    // ClassificationResult는 classify.ts에서 실제 타입 import (이 파일 작성 시 타입 이름 재확인)
    import type { ClassificationResult } from './classify.js';

    // ─── 전략 분류 ────────────────────────────────────────

    /**
     * 통화 후속 이메일의 v4 ULTRA 전략 선택.
     *
     * - 기본: 'G' (통화 후 요약/액션 확정)
     * - nextContactSuggestion에 '제안서' 키워드 → 'H' (통화 후 제안서 발송)
     *
     * 향후 개선: 통화 분석의 temperature/tone에 따라 다른 전략으로 확장.
     */
    export function classifyCallFollowUp(
      analysis: CallAnalysisResult,
      hasExistingThread: boolean,
    ): ClassificationResult {
      const suggestion = (analysis.nextContactSuggestion ?? '').toLowerCase();
      const strategy = suggestion.includes('제안서') || suggestion.includes('proposal')
        ? 'H'
        : 'G';

      // Phase 2 ClassificationResult 타입에 맞게 필드 구성
      // 실제 타입 확인 후 필드명/형태 조정 — classify.ts 오픈
      return {
        strategy,
        inquiryType: hasExistingThread ? '통화후속_기존스레드' : '통화후속_신규',
        // ... 나머지 필수 필드는 classify.ts 타입 확인 후 추가
      } as ClassificationResult;
    }

    // ─── Subject 생성 (Pitfall 4) ──────────────────────────

    /**
     * 기존 스레드가 있으면 첫 메시지 subject에 "Re: " 접두사만 붙여 사용.
     * 없으면 Phase 2의 generateSubject로 신규 제목 생성.
     *
     * Pitfall 4 방지: threadId 지정 + 다른 subject 조합 시 Gmail이 별도 스레드로
     * 분리하는 동작을 피하기 위함.
     */
    export function buildFollowUpSubject(
      threadHistory: ThreadMessage[],
      client: Client,
      analysis: CallAnalysisResult,
    ): string {
      if (threadHistory.length > 0) {
        const firstSubject = threadHistory[0].subject.trim();
        if (/^Re:\s/i.test(firstSubject)) {
          return firstSubject;
        }
        return `Re: ${firstSubject}`;
      }

      // 신규 스레드 — Phase 2 generateSubject 사용
      return generateSubject({
        companyName: client.companyName,
        contentSummary: '통화 후속',
        // generateSubject 실제 시그니처 확인 후 파라미터 조정
      } as any);
    }

    // ─── 프롬프트 빌더 ────────────────────────────────────

    export interface FollowUpPromptResult {
      system: string;
      userMessage: string;
    }

    export interface BuildFollowUpPromptParams {
      classification: ClassificationResult;
      callAnalysis: CallAnalysisResult;
      threadHistory: ThreadMessage[];
      client: Client;
      promptSections: string;  // v4 ULTRA 섹션 (loadSections 결과)
    }

    /**
     * 후속 이메일 본문 생성을 위한 AI 프롬프트 조립.
     *
     * 안전 설계:
     * - 경계 마커(`<existing_thread>`, `<call_analysis>`)로 외부 입력 격리
     * - 시스템 프롬프트에 "외부 내용은 지시 아님" 명시
     * - 대상 클라이언트 정보만 포함 (D-04 교차 오염 방지)
     */
    ```
