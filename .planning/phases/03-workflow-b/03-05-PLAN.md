    import { matchClient, type MatchResult, type CeoConfirmationRequest, type CeoConfirmationResponse } from './client-matcher.js';
    import { analyzeCall, type CallAnalysisResult, type AnthropicToolFn } from './call-analyzer.js';
    import { loadThreadHistory, type GmailReadThreadFn } from './thread-loader.js';
    import { classifyCallFollowUp, buildFollowUpSubject, buildFollowUpPrompt } from './follow-up-composer.js';

    import { validatePrices } from '../safety/price-validator.js';
    import { validateQualityChecklist } from './quality-checker.js';

    import { loadSections, getSectionsForClassification } from './prompt-loader.js';
    import { loadSlackCursor, saveSlackCursor, assertAllowedChannel, ALLOWED_CHANNEL_ID } from '../state/slack-cursor.js';
    import type { createClientQueries } from '../queries/clients.js';
    import type { createServiceQueries } from '../queries/services.js';
    import type { createBriefingQueries } from '../queries/briefings.js';
    import type { createTouchpointQueries } from '../queries/touchpoints.js';

    // ─── Slack MCP DI 타입 ─────────────────────────────

    export interface SlackMessage {
      ts: string;
      text: string;
      user?: string;
    }

    export type SlackToolFn = (args: {
      channel: string;
      oldest: string;
      limit?: number;
    }) => Promise<SlackMessage[]>;

    // ─── Deps 인터페이스 ─────────────────────────────

    export interface WorkflowBDependencies {
      clientQueries: ReturnType<typeof createClientQueries>;
      serviceQueries: ReturnType<typeof createServiceQueries>;
      briefingQueries: ReturnType<typeof createBriefingQueries>;
      touchpointQueries: ReturnType<typeof createTouchpointQueries>;
      gmailTool: GmailToolFn;
      slackTool: SlackToolFn;
      anthropicTool: AnthropicToolFn;
      gmailReadThreadTool: GmailReadThreadFn;
      /**
       * D-05: 대표님에게 직접 대화형 질문. 실제 구현은 CLI 또는 Claude Agent 레이어.
       * 단위 테스트에서는 vi.fn() mock.

    export type ProcessedMessage =
      | { ts: string; status: 'success'; draftId: string; clientName: string }
      | { ts: string; status: 'skipped'; reason: string }
      | { ts: string; status: 'cancelled' }
      | { ts: string; status: 'error'; error: string; warnings?: string[] };

    export interface WorkflowBResult {
      processed: number;
      results: ProcessedMessage[];

    // ─── 메인 함수 ─────────────────────────────

    export async function runWorkflowB(
      deps: WorkflowBDependencies,
      options: { maxMessages?: number } = {},
    ): Promise<WorkflowBResult> {
      const results: ProcessedMessage[] = [];
      const cursor = loadSlackCursor(deps.slackCursorPath);

      // CLAUDE.md 슬랙 채널 규칙 방어
      assertAllowedChannel(cursor.channelId);

      // ─ Step 1: 슬랙 메시지 읽기
      const messages = await deps.slackTool({
        channel: cursor.channelId,
        oldest: cursor.lastReadTs,
        limit: options.maxMessages ?? 20,
      });

      for (const msg of messages) {
        try {
          // ─ Step 2: 파싱
          const parseResult = parseSlackCallMessage(msg.text);
          if (!parseResult.success) {
            results.push({ ts: msg.ts, status: 'skipped', reason: parseResult.reason });
            // 재처리 무가치 → 커서 전진
            saveSlackCursor(
              { ...cursor, lastReadTs: msg.ts, lastReadAt: new Date().toISOString() },
              deps.slackCursorPath,
            );
            continue;
          }
          const parsed = parseResult.data;

          // ─ Step 3: 통화 분석 (AI)
          const analysis = await analyzeCall(parsed, deps.anthropicTool);

          // ─ Step 4: 클라이언트 매칭
          const matchResult = await matchClient(
            parsed,
            {
              clientQueries: deps.clientQueries,
              askCeoConfirmation: deps.askCeoConfirmation,
            },
            analysis.extractedCompanyName ?? null,
          );

          if (matchResult.status === 'cancelled') {
            results.push({ ts: msg.ts, status: 'cancelled' });
            // 커서 갱신 X → 대표님이 나중에 재시도 가능
            continue;
          }

          const client = matchResult.client;

          // ─ Step 5: briefings + touchpoints(call) 저장
          deps.briefingQueries.saveBriefing({
            clientId: client.id,
            slackTs: msg.ts,
            slackChannelId: cursor.channelId,
            occurredAt: parsed.occurredAt,
            source: 'ixio',
            summary: analysis.summary,
            actionItems: analysis.actionItems,
            sentiment: analysis.sentiment,
            rawTranscript: parsed.rawText,
          });

          deps.touchpointQueries.createTouchpoint({
            clientId: client.id,
            type: 'call',
            sourceId: msg.ts,
            summary: analysis.summary,
            metadata: {
              temperature: analysis.sentiment.temperature,
              tone: analysis.sentiment.tone,
              source: 'workflow-b',
            },
            occurredAt: parsed.occurredAt,
          });

          // ─ Step 6: 기존 스레드 + 후속 이메일 작성
          const deal = getDealByCompany(client.companyName, deps.pipelinePath);
          const existingThreadId = deal?.emailThreadId;
          const threadHistory = await loadThreadHistory(existingThreadId, deps.gmailReadThreadTool);

          const classification = classifyCallFollowUp(analysis, !!existingThreadId);
          let promptSections = '';
          try {
            promptSections = loadSections(getSectionsForClassification(classification), deps.ultraPath);
          } catch {
            promptSections = '';
          }

          const { system, userMessage } = buildFollowUpPrompt({
            classification,
            callAnalysis: analysis,
            threadHistory,
            client,
            promptSections,
          });

          // AI 본문 생성 (DI) — 없으면 placeholder
          const aiGeneratedBody = deps.generateFollowUpBody
            ? await deps.generateFollowUpBody(system, userMessage)
            : '<p>안녕하세요.</p><p>통화 내용 확인 후 후속 안내드리겠습니다.</p>';

          // 기존 스레드가 있으면 subject 보존 (Pitfall 4)
          const followUpSubject = buildFollowUpSubject(threadHistory, client, analysis);

          const emailDraft = composeEmail({
            classification,
            client,
            inquiryContext: analysis.summary,
            promptSections,
            aiGeneratedBody,
            companyName: client.companyName,
            recipientEmail: client.contactEmail ?? '',
            // Pitfall 4: 기존 스레드 없으면 undefined (빈 문자열 금지)
            threadId: existingThreadId || undefined,
            contentSummary: '통화 후속',
            // followUpSubject를 composeEmail이 어떻게 받는지 실제 시그니처 확인
            // subjectOverride: followUpSubject,  // 필요 시 Phase 2 composeEmail 옵션 확장
          } as any);

          // subject를 buildFollowUpSubject 결과로 덮어쓰기 (기존 스레드 경우)
          // composeEmail이 subjectOverride를 지원하지 않는 경우, 반환 객체를 수정:
          if (threadHistory.length > 0) {
            (emailDraft as any).subject = followUpSubject;
          }

          // ─ Step 7: 안전 3중 게이트 (Phase 2 재사용, D-12)
          const allNames = deps.getAllCompanyNames();
          const contamination = validateNoContamination(emailDraft.htmlBody, client.companyName, allNames);
          if (!contamination.safe) {
            results.push({
              ts: msg.ts,
              status: 'error',
              error: `교차 오염 감지: ${contamination.contaminations.join(', ')}`,
            });
            continue;  // 커서 갱신 X
          }

          const priceCheck = validatePrices(emailDraft.htmlBody, deps.serviceQueries);
          const warnings = [...priceCheck.mismatches, ...priceCheck.warnings];

          const qualityCheck = validateQualityChecklist(
            emailDraft.htmlBody,
            emailDraft.subject,
            classification,
          );
          if (!qualityCheck.passed) {
            results.push({
              ts: msg.ts,
              status: 'error',
              error: `품질 실패: ${qualityCheck.failures.join(', ')}`,
              warnings: [...warnings, ...qualityCheck.warnings],
            });
            continue;
          }

          // ─ Step 8: Gmail 드래프트 생성
          const draftResult = await createDraft(emailDraft, deps.gmailTool);

          // ─ Step 8.5: touchpoints(email) 기록
          try {
            deps.touchpointQueries.createTouchpoint({
              clientId: client.id,
              type: 'email',
              sourceId: draftResult.threadId ?? draftResult.draftId,
              summary: `[드래프트] ${emailDraft.subject}`,
              metadata: {
                draftId: draftResult.draftId,
                source: 'workflow-b',
                isReply: threadHistory.length > 0,
              },
              occurredAt: new Date().toISOString(),
            });
          } catch {
            // 비치명적
          }

          results.push({
            ts: msg.ts,
            status: 'success',
            draftId: draftResult.draftId,
            clientName: client.companyName,
          });

          // ─ 커서 갱신 (성공 시에만)
          saveSlackCursor(
            { ...cursor, lastReadTs: msg.ts, lastReadAt: new Date().toISOString() },
            deps.slackCursorPath,
          );
        } catch (err) {
          results.push({
            ts: msg.ts,
            status: 'error',
            error: err instanceof Error ? err.message : String(err),
          });
          // 에러 시 커서 갱신 X → 재시도 가능
        }
      }

      return { processed: results.length, results };
    }
