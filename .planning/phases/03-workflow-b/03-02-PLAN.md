    **1) `src/workflow/phone-normalizer.ts`:**
    ```typescript
    /**
     * 전화번호를 단일 정규 형식(11자리 숫자)으로 통일.
     *
     * 변환 규칙:
     *  - 모든 비숫자 문자 제거 (공백/하이픈/괄호/+ 등)
     *  - '82'로 시작하면 국가코드로 간주하여 '0' + 나머지로 치환
     *
     * 예:
     *  '010-3690-6867'   → '01036906867'
     *  '+82 10-3690-6867' → '01036906867'
     *  '+821036906867'    → '01036906867'
     *  '02-1234-5678'    → '0212345678'
     *
     * 단일 유틸: DB 저장 / DB 검색 / 파서 출력 모두 이 함수를 경유해야 한다 (Pitfall 1 방지).
     */
    export function normalizePhone(raw: string): string {
      if (!raw) return '';
      const digits = raw.replace(/\D/g, '');
      if (digits.startsWith('82')) return '0' + digits.slice(2);
      return digits;
    }
    ```

    ```typescript
    import { z } from 'zod';
    import { normalizePhone } from './phone-normalizer.js';

    export const ParsedCallSchema = z.object({
      source: z.literal('ixio'),
      shareType: z.enum(['대화', '통화']),
      occurredAt: z.string(),
      contact: z.object({
        phone: z.string().nullable(),
        companyName: z.string().nullable(),
        contactName: z.string().nullable(),
        contactTitle: z.string().nullable(),
      }),
      mainPoints: z.array(z.string()).nullable(),
      details: z.array(z.string()).nullable(),
      transcript: z.array(z.object({
        speaker: z.enum(['me', 'other']),
        text: z.string(),
      })).nullable(),
      rawText: z.string(),
    });
    export type ParsedCall = z.infer<typeof ParsedCallSchema>;

    export type ParseResult =
      | { success: true; data: ParsedCall }
      | { success: false; reason: string };

    const HEADER_RE = /^익시오에서 .+?님이 공유한 (대화|통화) 내용입니다\.$/m;
    const DATE_RE = /(\d{4})\. (\d{2})\. (\d{2}) \([월화수목금토일]요일\) (오전|오후) (\d{1,2})시 (\d{2})분/;
    const CONTACT_WITH_NAME_RE = /^연락처 (.+?)\((\d{9,11})\)님과의 (통화|대화)$/m;
    const CONTACT_PHONE_ONLY_RE = /^연락처 (\d{9,11})님과의 (통화|대화)$/m;
    const SECTION_HEADERS = ['[주요 내용]', '[상세 내용]', '[대화 내용]'] as const;

    export function parseSlackCallMessage(text: string): ParseResult {
      const headerMatch = text.match(HEADER_RE);
      if (!headerMatch) return { success: false, reason: 'not an ixio message' };
      const shareType = headerMatch[1] as '대화' | '통화';

      const dateMatch = text.match(DATE_RE);
      if (!dateMatch) return { success: false, reason: 'date format unrecognized' };
      const occurredAt = buildIsoDate(dateMatch);

      const contact = parseContact(text);
      const sections = splitSections(text);

      const parsed: ParsedCall = {
        source: 'ixio',
        shareType,
        occurredAt,
        contact,
        mainPoints: sections['[주요 내용]'] ?? null,
        details: sections['[상세 내용]'] ?? null,
        transcript: sections['[대화 내용]']
          ? parseTranscript(sections['[대화 내용]'].join('\n'))
          : null,
        rawText: text,
      };

      // zod 최종 검증 (스키마 불일치 시 throw)
      return { success: true, data: ParsedCallSchema.parse(parsed) };
    }

    function buildIsoDate(m: RegExpMatchArray): string {
      const [, year, month, day, period, hourStr, minute] = m;
      let hour = parseInt(hourStr, 10);
      if (period === '오후' && hour < 12) hour += 12;
      if (period === '오전' && hour === 12) hour = 0;
      return `${year}-${month}-${day}T${String(hour).padStart(2, '0')}:${minute}:00+09:00`;
    }

    function parseContact(text: string): ParsedCall['contact'] {
      const named = text.match(CONTACT_WITH_NAME_RE);
      if (named) {
        // "퍼즐) 동반성장위원회 이의선 위원" 같은 형태
        // ')'로 prefix(퍼즐) 제거 후, 공백 split으로 회사/이름/직함 분리
        const afterParen = named[1].includes(')')
          ? named[1].split(')').slice(1).join(')').trim()
          : named[1].trim();
        const words = afterParen.split(/\s+/).filter(Boolean);

        let companyName: string | null = null;
        let contactName: string | null = null;
        let contactTitle: string | null = null;

        if (words.length >= 3) {
          companyName = words[0];
          contactTitle = words[words.length - 1];
          contactName = words.slice(1, -1).join(' ');
        } else if (words.length === 2) {
          companyName = words[0];
          contactName = words[1];
        } else if (words.length === 1) {
          contactName = words[0];
        }

        return {
          phone: normalizePhone(named[2]),
          companyName,
          contactName,
          contactTitle,
        };
      }
      const phoneOnly = text.match(CONTACT_PHONE_ONLY_RE);
      if (phoneOnly) {
        return {
          phone: normalizePhone(phoneOnly[1]),
          companyName: null,
          contactName: null,
          contactTitle: null,
        };
      }
      return { phone: null, companyName: null, contactName: null, contactTitle: null };
    }

    function splitSections(text: string): Record<string, string[]> {
      const result: Record<string, string[]> = {};
      for (const header of SECTION_HEADERS) {
        const idx = text.indexOf(header);
        if (idx === -1) continue;
        const nextHeaderIdx = SECTION_HEADERS
          .filter(h => h !== header)
          .map(h => text.indexOf(h, idx + header.length))
          .filter(i => i !== -1)
          .sort((a, b) => a - b)[0] ?? text.length;
        const block = text.slice(idx + header.length, nextHeaderIdx).trim();
        result[header] = block.split('\n').map(l => l.trim()).filter(l => l.length > 0);
      }
      return result;
    }

    function parseTranscript(text: string): Array<{ speaker: 'me' | 'other'; text: string }> {
      const lines = text.split('\n').filter(l => l.trim().length > 0);
      const turns: Array<{ speaker: 'me' | 'other'; text: string }> = [];
      for (const line of lines) {
        const match = line.match(/^(나|상대방):\s*(.*)$/);
        if (match) {
          turns.push({
            speaker: match[1] === '나' ? 'me' : 'other',
            text: match[2],
          });
        } else if (turns.length > 0) {
          turns[turns.length - 1].text += ' ' + line;
        }
      }
      return turns;
    }
    ```

    ```typescript
        getClientByPhone(phone: string) {
          if (!phone) return undefined;
          const normalized = normalizePhone(phone);
          if (!normalized) return undefined;
          return db.select().from(clients)
            .where(eq(clients.contactPhone, normalized))
            .get();
        },

        /**
         * D-04: 이메일로 단일 클라이언트 조회.
         * 대소문자 무시 (입력을 lowercase로 변환하여 조회).
         */
        getClientByEmail(email: string) {
          if (!email) return undefined;
          const lower = email.trim().toLowerCase();
          // SQLite COLLATE NOCASE 또는 Drizzle sql raw 사용. 단순 접근: lower로 저장 가정 + lower 조회.
          // 안전한 대안: sql`lower(${clients.contactEmail}) = ${lower}` — 두 방식 중 기존 Phase 1/2 패턴 확인 후 선택.
          return db.select().from(clients)
            .where(eq(clients.contactEmail, lower))
            .get();
        },

        /**
         * D-03: 통화로 확인된 전화번호를 기존 client에 저장(학습).
         * 반드시 normalizePhone을 거쳐 저장 (Pitfall 1).
         * 존재하지 않는 회사명이면 no-op (오케스트레이터가 매칭 후 호출하므로 존재 전제).
         */
        savePhoneLearning(companyName: string, phone: string): void {
          const normalized = normalizePhone(phone);
          if (!normalized) return;
          db.update(clients)
            .set({
              contactPhone: normalized,
              updatedAt: new Date().toISOString(),
            })
            .where(eq(clients.companyName, companyName))
            .run();
        },

        /**
         * 모든 회사명 목록 (fuzzy 매칭 전용).
         * D-04 예외: 회사명만 반환 — Client 전체 정보 노출 금지.
         * 이 함수의 출력은 client-matcher.ts의 fuzzyMatchCompanyName에만 전달된다.
         */
        getAllCompanyNames(): string[] {
          return db.select({ companyName: clients.companyName })
            .from(clients)
            .all()
            .map(r => r.companyName);
        },
    ```

    ```typescript
    export const COMPANY_BLACKLIST = ['퍼즐', 'puzl', '퍼즐코퍼레이션', 'puzzle corporation'];

    export interface MatchCandidate {
      companyName: string;
      score: number;
      reason: 'exact' | 'prefix' | 'contains';

      | { status: 'matched'; client: Client; method: 'phone_exact' | 'name_exact' | 'name_fuzzy' | 'ceo_select'; confidence: number }
      | { status: 'new_client'; client: Client; method: 'ceo_new' }

    export function normalizeKoreanCompany(name: string): string {
      return name
        .toLowerCase()
        .replace(/\s+/g, '')
        .replace(/[()[\]{}/,.\-]/g, '')
        .replace(/(주식회사|㈜|\(주\)|주\))/g, '')
        .replace(/(코퍼레이션|corp|corporation|inc|ltd)/gi, '');
    }

    function isBlacklisted(name: string): boolean {
      const n = normalizeKoreanCompany(name);
      return COMPANY_BLACKLIST.some(b => normalizeKoreanCompany(b) === n || n.includes(normalizeKoreanCompany(b)));
    }

    export function fuzzyMatchCompanyName(
      extracted: string,
      allCompanyNames: string[],
    ): MatchCandidate[] {
      const needle = normalizeKoreanCompany(extracted);
      if (!needle) return [];
      const candidates: MatchCandidate[] = [];

      for (const name of allCompanyNames) {
        if (isBlacklisted(name)) continue;
        const hay = normalizeKoreanCompany(name);
        if (!hay) continue;

        if (hay === needle) {
          candidates.push({ companyName: name, score: 1.0, reason: 'exact' });
        } else if (hay.startsWith(needle) || needle.startsWith(hay)) {
          const longer = Math.max(hay.length, needle.length);
          const shorter = Math.min(hay.length, needle.length);
          candidates.push({
            companyName: name,
            score: 0.75 + (shorter / longer) * 0.20,
            reason: 'prefix',
          });
        } else if (hay.includes(needle) || needle.includes(hay)) {
          candidates.push({ companyName: name, score: 0.70, reason: 'contains' });
        }
      }

      return candidates.sort((a, b) => b.score - a.score);
    }

    export async function matchClient(
      parsed: ParsedCall,
      deps: ClientMatcherDeps,
      extractedCompanyName: string | null,
    ): Promise<MatchResult> {
      // ─ Step 1: 전화번호 정확 매칭
      if (parsed.contact.phone) {
        const byPhone = deps.clientQueries.getClientByPhone(parsed.contact.phone);
        if (byPhone) {
          return { status: 'matched', client: byPhone, method: 'phone_exact', confidence: 1.0 };
        }
      }

      // ─ Step 2: 회사명 기반 매칭
      // 자사 블랙리스트 필터링 (Pitfall 6)
      let nameToMatch = parsed.contact.companyName ?? extractedCompanyName;
      if (nameToMatch && isBlacklisted(nameToMatch)) {
        nameToMatch = null;
      }

      if (nameToMatch) {
        // 2a. 정확 매칭
        const exact = deps.clientQueries.getClient(nameToMatch);
        if (exact) {
          if (parsed.contact.phone) {
            deps.clientQueries.savePhoneLearning(exact.companyName, parsed.contact.phone);
          }
          return { status: 'matched', client: exact, method: 'name_exact', confidence: 1.0 };
        }

        // 2b. 퍼지 매칭
        const allNames = deps.clientQueries.getAllCompanyNames();
        const candidates = fuzzyMatchCompanyName(nameToMatch, allNames);
        const top = candidates[0];

        // Pitfall 3 방지: 자동 통과는 엄격 AND 조건
        const autoPassOk =
          top !== undefined &&
          top.score >= 0.95 &&
          candidates.length === 1 &&
          (top.reason === 'exact' || top.reason === 'prefix');

        if (autoPassOk) {
          const client = deps.clientQueries.getClient(top.companyName);
          if (client) {
            if (parsed.contact.phone) {
              deps.clientQueries.savePhoneLearning(client.companyName, parsed.contact.phone);
            }
            return { status: 'matched', client, method: 'name_fuzzy', confidence: top.score };
          }
        }
      }

      const candidatesForCeo = nameToMatch
        ? fuzzyMatchCompanyName(nameToMatch, deps.clientQueries.getAllCompanyNames())
            .slice(0, 5)
            .map(c => {
              const client = deps.clientQueries.getClient(c.companyName);

          phone: parsed.contact.phone,
          companyName: nameToMatch,
          contactName: parsed.contact.contactName,
        },
        candidates: candidatesForCeo,
      });

        return { status: 'cancelled' };
      }

      if (response.action === 'select' && response.selectedClientId != null) {
        const client = deps.clientQueries.getClientById(response.selectedClientId);
        if (!client) throw new Error(`선택된 클라이언트 ID가 존재하지 않음: ${response.selectedClientId}`);
        if (parsed.contact.phone) {
          deps.clientQueries.savePhoneLearning(client.companyName, parsed.contact.phone);
        }
        return { status: 'matched', client, method: 'ceo_select', confidence: 1.0 };
      }

      if (response.action === 'new_client' && response.newClientName) {
        deps.clientQueries.upsertClient({
          companyName: response.newClientName,
          contactName: parsed.contact.contactName,
          contactPhone: parsed.contact.phone ? normalizePhone(parsed.contact.phone) : null,

      }

      throw new Error('대표님 응답 파싱 실패: ' + JSON.stringify(response));
    }
    ```
