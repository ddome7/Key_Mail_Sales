    ```typescript
      const mainPointsStr = parsed.mainPoints?.length

        : '(대화 전문 없음)';

      const userMessage = [
        `[메타데이터]`,
        `- 발생 시각 (기준일): ${parsed.occurredAt}`,
        `- 상대방 (파서 추출): ${parsed.contact.companyName ?? '(회사 미상)'} / ${parsed.contact.contactName ?? '(이름 미상)'}`,

        '',
        '[주요 내용]',
        mainPointsStr,
        '',
        '[상세 내용]',

        '</call_transcript>',

        '위 내용을 분석해서 record_call_analysis 도구를 호출하세요.',
      ].join('\n');

      return { system, userMessage };
    }
    ```

    ```typescript
     * @param anthropicTool - DI된 Anthropic tool_use 호출 함수
     * @returns 구조화된 분석 결과 (zod 검증 완료)
     * @throws tool_use 응답 누락, 스키마 불일치, 빈 응답 등
     *
     * 자사 필터: AI가 반환한 extractedCompanyName이 자사 이름이면 null로 치환 (Pitfall 6).
     */
    export async function analyzeCall(
      parsed: ParsedCall,
      anthropicTool: AnthropicToolFn,
    ): Promise<CallAnalysisResult> {
      const { system, userMessage } = buildCallAnalysisPrompt(parsed);

      const response = await anthropicTool({
        system,
        userMessage,
        tool: CALL_ANALYSIS_TOOL_DEFINITION,
        forceTool: true,
      });

      // tool_use 블록 추출
      const toolBlock = response.content.find(
        (b): b is Extract<typeof b, { type: 'tool_use' }> =>
          b.type === 'tool_use' && (b as any).name === CALL_ANALYSIS_TOOL_NAME,
      );

      if (!toolBlock) {
        throw new Error(
          `analyzeCall: tool_use 응답 누락. 기대: ${CALL_ANALYSIS_TOOL_NAME}, 실제 content types: ${response.content.map(b => b.type).join(',')}`,
        );
      }

      // zod 검증 (스키마 불일치 시 throw)
      const parsedResult = CallAnalysisSchema.parse(toolBlock.input);

      // Pitfall 6: 자사 이름 필터
      const filteredCompany = parsedResult.extractedCompanyName
        && isCompanyBlacklisted(parsedResult.extractedCompanyName)
        ? null
        : parsedResult.extractedCompanyName ?? null;

      return {
        ...parsedResult,
        extractedCompanyName: filteredCompany,
      };
    }
    ```
