    // src/workflow/cadence-report.ts
    /**
     * Phase 4 D-06: Workflow C 실행 결과 리포트 렌더러 (순수 함수).
     * workflow-c.ts에서 분리 — 포맷팅 책임만 담당, DB/네트워크 접근 0.
     */
    import type { WorkflowCBatchResult, WorkflowCResult } from './workflow-c.js';

    const STATUS_LABELS: Record<string, string> = {
      success: '✅ 드래프트',
      no_action: '⏭️ 스킵',
      error: '❌ 오류',
    };

    const REASON_STATUS_LABELS: Record<string, string> = {
      recent_inbound: '⏭️ 답장 수신',
      no_stage_due: '⏸️ 미도래',
      paused: '⏸️ 수동 차단',
      dormant: '🌙 휴면',
      no_last_outbound: '❔ 접점 없음',
      skipped: '❔ 미확인',
    };

    function statusLabelForRow(row: WorkflowCBatchResult['rows'][number]): string {
      if (row.status === 'success') {
        return row.note === '휴면 전환 확인 필요' ? '⚠️ 휴면 확인' : '✅ 드래프트';
      }
      if (row.status === 'error') return '❌ 오류';
      // no_action — note에 한국어 사유 매핑되어있음 (04-05 REASON_LABELS). 그대로 사용.
      return row.note ? `⏭️ ${row.note}` : '⏭️ 스킵';
    }

    export function renderCadenceReport(
      batch: WorkflowCBatchResult,
      meta?: { now?: Date; maxDrafts?: number },
    ): string {
      const now = (meta?.now ?? new Date()).toISOString().slice(0, 10);
      const header = [
        `## Workflow C 실행 결과 — ${now}`,
        '',
        `**처리:** ${batch.processed}건 검토 / ${batch.drafted}건 드래프트 생성 / ${batch.skipped}건 스킵 / ${batch.errors}건 오류`,
        '',
      ];

      const tableHeader = [
        '| 회사명 | 마지막 접점 | 경과 | 다음 케이던스 | 드래프트 ID | 상태 |',
        '|--------|-------------|------|---------------|-------------|------|',
      ];

      const tableRows = batch.rows.map((r) => {
        const cols = [
          r.companyName,
          r.lastOutboundAt ?? '—',
          r.elapsedDays == null ? '—' : `${r.elapsedDays}일`,
          r.nextCadence ?? '—',
          r.draftId ?? '—',
          statusLabelForRow(r),
        ];
        return `| ${cols.join(' | ')} |`;
      });

      const sections: string[] = [...header, ...tableHeader, ...tableRows];

      if (batch.pendingBeyondCap > 0) {
        sections.push(
          '',
          '### 대기 중',
          `처리 상한(${meta?.maxDrafts ?? 10}건)을 초과한 ${batch.pendingBeyondCap}건이 남았습니다. 재실행이 필요합니다.`,
        );
      }

      if (batch.dormantCandidates.length > 0) {
        sections.push(
          '',
          '### 🌙 휴면 전환 확인 필요 (D-17)',
          '',
          '아래 클라이언트는 30일 이상 미응답으로 STAGE 5 (Dormant) 전환 대상입니다. "마지막 터치" 드래프트를 작성해 두었으니 대표님이 확인해 주시고 전환 여부를 알려주세요.',
          '',
        );
        batch.dormantCandidates.forEach((c, i) => {
          sections.push(
            `${i + 1}. **${c.companyName}** — 마지막 드래프트 \`${c.draftId}\``,
            `   - 대표님 결정 필요: "전환해" / "유지해"`,
          );
        });
      }

      return sections.join('\n');
    }

    export function renderSingleResult(result: WorkflowCResult): string {
      if (result.status === 'success') {
        const lines = [`✅ ${result.companyName} 드래프트 생성 (${result.draftId})`];
        lines.push(`   - 단계: ${result.stage.ordinal}차 (Day ${result.stage.dayMilestone}) / ${result.stage.contextType}`);
        if (result.dormantPending) lines.push('   - ⚠️ 휴면 전환 확인 필요');
        if (result.warnings && result.warnings.length > 0) {
          lines.push(`   - 경고: ${result.warnings.join(' · ')}`);
        }
        return lines.join('\n');
      }
      if (result.status === 'error') {
        return `❌ ${result.companyName}: ${result.error}`;
      }
      const reasonLabel = REASON_STATUS_LABELS[result.reason] ?? result.reason;
      return `${reasonLabel} ${result.companyName}`;
    }
    ```

     *   --confirm-dormant     휴면 전환 대화형 확인 활성화 (readline y/n)
     *   --help                이 도움말
     *
     * 사용 예:
     *   npx tsx scripts/run-workflow-c.ts
     *   npx tsx scripts/run-workflow-c.ts --company 동반성장위원회
     *   npx tsx scripts/run-workflow-c.ts --max 5 --confirm-dormant
     *
     * 실행 모드: on-demand. 주기 실행 API는 사용하지 않는다 (D-01).
     */

    import Database from 'better-sqlite3';
    import { drizzle } from 'drizzle-orm/better-sqlite3';
    import * as readline from 'node:readline/promises';
    import { stdin, stdout } from 'node:process';

    import { createClientQueries } from '../src/queries/clients.js';
    import { createTouchpointQueries } from '../src/queries/touchpoints.js';
    import { createBriefingQueries } from '../src/queries/briefings.js';
    import { createServiceQueries } from '../src/queries/services.js';
    import { runWorkflowC, type WorkflowCDependencies, type WorkflowCBatchResult } from '../src/workflow/workflow-c.js';
    import { renderCadenceReport, renderSingleResult } from '../src/workflow/cadence-report.js';

    interface CliFlags {
      company?: string;
      max: number;
      confirmDormant: boolean;
      help: boolean;
    }

    function parseFlags(argv: string[]): CliFlags {
      const flags: CliFlags = { max: 10, confirmDormant: false, help: false };
      for (let i = 0; i < argv.length; i++) {
        const a = argv[i];
        if (a === '--company') { flags.company = argv[++i]; }
        else if (a === '--max') { flags.max = parseInt(argv[++i], 10) || 10; }
        else if (a === '--confirm-dormant') { flags.confirmDormant = true; }
        else if (a === '--help' || a === '-h') { flags.help = true; }
      }
      return flags;
    }

    function printHelp(): void {
      console.log(`
    Workflow C — 팔로업 자동화 (Phase 4)

    사용법:
      npx tsx scripts/run-workflow-c.ts [옵션]

    옵션:
      --company <이름>        특정 클라이언트 1건만 처리 (모드 B)
      --max <N>              전수 스캔 시 처리 상한 (기본 10)
      --confirm-dormant      휴면 전환 대화형 확인 활성화
      --help                 이 도움말 출력

    실행 예:
      npx tsx scripts/run-workflow-c.ts
      npx tsx scripts/run-workflow-c.ts --company 동반성장위원회
      npx tsx scripts/run-workflow-c.ts --max 5 --confirm-dormant
    `.trim());
    }

    async function main(): Promise<void> {
      const flags = parseFlags(process.argv.slice(2));
      if (flags.help) { printHelp(); return; }

      // DB 연결
      const sqlite = new Database('data/sales.db');
      const db = drizzle(sqlite);

      const clientQueries = createClientQueries(db);
      const touchpointQueries = createTouchpointQueries(db);
      const briefingQueries = createBriefingQueries(db);
      const serviceQueries = createServiceQueries(db);

      // Gmail MCP 도구는 Claude Code 세션이 실행할 때만 유효.
      // tsx 스크립트 단독 실행 모드에서는 stub을 주입 — 드래프트는 실제로 생성되지 않고
      // workflow-c 내부 로직(cadence 판정, prompt 조립, 안전 게이트)만 검증됨.
      // 실제 Gmail 드래프트 생성은 Claude Code 세션에서 04-07 UAT 스크립트를 호출할 때만.
      const gmailToolStub: WorkflowCDependencies['gmailTool'] = async () => ({
        draftId: 'stub-draft-id', threadId: undefined, messageId: undefined,
      } as any);
      const gmailReadThreadStub: WorkflowCDependencies['gmailReadThreadTool'] = async () => [];

      // readline 기반 askCeoDormantConfirmation (opt-in)
      let askCeoDormantConfirmation: WorkflowCDependencies['askCeoDormantConfirmation'] = undefined;
      let rl: readline.Interface | undefined;
      if (flags.confirmDormant) {
        rl = readline.createInterface({ input: stdin, output: stdout });
        askCeoDormantConfirmation = async (client, draftId) => {
          console.log(`\n🌙 ${client.companyName} 휴면 전환 확인`);
          console.log(`   마지막 드래프트: ${draftId}`);
          const ans = (await rl!.question('   전환하시겠어요? [y/n]: ')).trim().toLowerCase();
          return ans === 'y' || ans === 'yes' ? 'confirm' : 'keep_active';
        };
      }

      const deps: WorkflowCDependencies = {
        clientQueries,
        serviceQueries,
        touchpointQueries,
        briefingQueries,
        gmailTool: gmailToolStub,
        gmailReadThreadTool: gmailReadThreadStub,
        getAllCompanyNames: () => clientQueries.getAllCompanyNames(),
        askCeoDormantConfirmation,
        // generateCadenceFollowUpBody / classifyContextWithAI 미주입
        // → Claude Code 모드: placeholder body + deterministic classifier
      };

      try {
        const result = await runWorkflowC(deps, {
          maxDrafts: flags.max,
          companyName: flags.company,
        });

        if ('rows' in result) {
          console.log(renderCadenceReport(result, { maxDrafts: flags.max }));
        } else {
          console.log(renderSingleResult(result));
        }
      } finally {
        rl?.close();
        sqlite.close();
      }
    }

    main().catch((err) => {
      console.error('Workflow C 실행 실패:', err);
      process.exit(1);
    });
    ```

<pitfalls>
## Plan-specific Gotchas

1. **CLI 단독 실행 시 Gmail 드래프트 생성 불가:** `npx tsx` 프로세스는 Claude Code 세션의 MCP 도구에 접근할 수 없다. 따라서 이 플랜의 CLI는 "로직 파이프라인 검증" 용도이며 실제 드래프트 생성은 04-07 UAT 스크립트가 Claude Code 세션 내부에서 호출될 때만 동작한다. CLI 주석과 --help 문구에 이 제약을 **명시**(대표님 비개발자이므로 혼동 방지).

2. **Gmail stub의 draftId:** stub은 'stub-draft-id' 같은 고정 문자열 반환. 이 상태로 touchpoint가 기록되어도 운영 DB에 stub draftId가 남게 됨. 대표님에게 혼동 → 해결: **CLI 실행 시 `--dry-run` 기본 모드** 개념으로 touchpoint 기록을 **건너뛰도록** `touchpointQueries`를 wrap하거나, 최소한 stub 실행 시 경고 메시지를 출력. **가장 간단한 해결:** 04-06 CLI는 워낙 "이것은 smoke test" 용도이므로 `--company` 없는 모드 A는 기본적으로 "dry-run 리포트만 표시"를 기본값으로 하고, 실제 DB 쓰기는 Claude Code 세션(UAT) 전용으로 두어 안전한 분리. 이 방침을 CLI 도움말에 명시.

3. **readline 기반 confirm-dormant:** `readline/promises`는 Node 18+ 지원. tsx가 이 API를 그대로 받음. 테스트 환경에서는 flags.confirmDormant=false로 호출하여 readline 인스턴스 생성 회피.

4. **더 좋은 방법 (선택):** 복잡성을 낮추려면 CLI를 전적으로 "실행 자체를 Claude Code 세션이 하도록" 설계하고, 04-06은 그냥 `runWorkflowC` 호출 + `renderCadenceReport` 출력으로 단순화. 04-07이 DI 조립의 본체가 되도록. 이 플랜은 **전자**(CLI가 DI 조립)를 선택하되 Gmail 부분만 stub이라는 타협.

5. **D-01 재확인:** CLI 파일 최종 저장 직전 grep으로 금지 단어 0건 확인. 주석의 "주기 실행"은 "(금지 — on-demand only)" 표기 또는 "on-demand 모드"로 대체.

6. **DB 경로 하드코딩:** `'data/sales.db'` 는 프로젝트 루트 기준. tsx 실행 시 cwd가 프로젝트 루트여야 함. 아니면 `path.resolve(__dirname, '../data/sales.db')` 로 절대 경로 해결. ESM에서 `__dirname`이 없으므로 `new URL('../data/sales.db', import.meta.url).pathname` 패턴.

7. **cadence-report 렌더러 내 markdown 테이블 escape:** 회사명에 `|` 문자가 있을 가능성 극히 낮지만 방어적으로 `.replace(/\|/g, '\\|')` 적용 고려 (optional).
