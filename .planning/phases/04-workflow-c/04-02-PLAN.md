    // src/workflow/cadence.ts
    /**
     * Phase 4 순수 함수 레이어 — cadence 판정 + deterministic classifier.
     * DB/네트워크 의존성 0. v4 ULTRA §15 규칙 구현.
     */

    export type CadenceContextType =
      | 'proposal'
      | 'contract'
      | 'meeting_qna'
      | 'quote_revision'
      | 'influencer_outreach'
      | 'general';

    /**
     * context_type별 Day 경과 milestone 시퀀스.
     * v4 ULTRA §15 L871-878 [상황별 팔로업 변형] + CONTEXT.md D-12 해석.
     *
     * D-12 해석 주의: v4 ULTRA 본문은 proposal의 Day 30을 명시하지 않으나
     * CONTEXT.md는 "proposal: Day 3 → 7 → 14 → 30"으로 결정. FLLW-04(Dormant
     * 전환)과 일관성을 위해 30일 마지막 stage 포함.
     */
    export const CADENCE_MATRIX: Record<CadenceContextType, readonly number[]> = {
      general:              [3, 7, 14, 30],
      proposal:             [3, 7, 14, 30],
      contract:             [2, 5],
      meeting_qna:          [2, 5],
      quote_revision:       [3, 7],
      influencer_outreach:  [2, 5, 10],
    } as const;

    /** Deterministic classifier용 키워드 그룹 — 우선순위 순서 */
    const CONTEXT_KEYWORDS: ReadonlyArray<readonly [CadenceContextType, ReadonlyArray<string>]> = [
      ['contract',            ['계약서', '계약 체결', 'contract', 'NDA', '법인인감', '사인']],
      ['quote_revision',      ['견적 수정', '견적서 수정', '수정 견적', '재견적', '단가 조정']],
      ['proposal',            ['제안서', '기획안', '캠페인 제안', 'proposal', '제안 드립니다', '제안드립니다']],
      ['meeting_qna',         ['미팅 감사', '미팅 후', 'QnA', '질의응답', '답변 드립니다', '미팅에서 논의']],
      ['influencer_outreach', ['크리에이터', '인플루언서', '협업 제안', 'UGC', '콘텐츠 제작 의뢰']],
    ];

    export function classifyCadenceContextDeterministic(
      outboundBody: string,
      outboundSubject: string,
    ): CadenceContextType | null {
      const haystack = `${outboundSubject}\n${outboundBody}`;
      for (const [type, keywords] of CONTEXT_KEYWORDS) {
        if (keywords.some((kw) => haystack.includes(kw))) return type;
      }
      return null;
    }

    export interface CadenceStageResult {
      stageIndex: number;
      dayMilestone: number;
      ordinal: number;
      toneIntent: 'confirm' | 'add_value' | 'alternative' | 'relationship';
      isFinalTouch: boolean;
      toneLevel: 1 | 2 | 3 | 4 | 5;
    }

    export const TONE_INTENT_LABELS: Record<CadenceStageResult['toneIntent'], string> = {
      confirm:      '확인 여부만 물어보기',
      add_value:    '새로운 가치 추가 — 답장할 이유 제공',
      alternative:  '대안 제시 — 다른 시나리오 / 작은 규모',
      relationship: '관계 유지 전환 — 압박 제로, 따뜻하게',
    };

    /**
     * @param lastOutboundAt ISO 8601 — 마지막 outbound 시점
     * @param now 현재 시점 (테스트에서 DI)
     * @param contextType CADENCE_MATRIX 키
     * @param executedStageIndices 이미 실행된 stageIndex 배열 (D-13 재사용)
     */
    export function computeCadenceStage(
      lastOutboundAt: string,
      now: Date,
      contextType: CadenceContextType,
      executedStageIndices: number[] = [],
    ): CadenceStageResult | null {
      const lastMs = new Date(lastOutboundAt).getTime();
      const nowMs = now.getTime();
      if (Number.isNaN(lastMs)) return null;

      const elapsedDays = Math.floor((nowMs - lastMs) / (24 * 60 * 60 * 1000));
      if (elapsedDays < 0) return null;

      const milestones = CADENCE_MATRIX[contextType];
      let targetIdx = -1;
      for (let i = 0; i < milestones.length; i++) {
        if (elapsedDays >= milestones[i] && !executedStageIndices.includes(i)) {
          targetIdx = i;
        }
      }
      if (targetIdx === -1) return null;

      const ordinal = targetIdx + 1;
      const isFinal = targetIdx === milestones.length - 1;

      const toneIntent: CadenceStageResult['toneIntent'] =
        isFinal ? 'relationship'
        : ordinal === 1 ? 'confirm'
        : ordinal === 2 ? 'add_value'
        : 'alternative';

      const toneLevel: 1 | 2 | 3 | 4 | 5 =
        toneIntent === 'relationship' ? 2
        : toneIntent === 'confirm'    ? 2
        : toneIntent === 'add_value'  ? 2
        : 3;

      return {
        stageIndex: targetIdx,
        dayMilestone: milestones[targetIdx],
        ordinal,
        toneIntent,
        isFinalTouch: isFinal,
        toneLevel,
      };
    }
    ```

    // src/workflow/inbound-detector.ts
    /**
     * Phase 4: 스레드 inbound 감지 순수 함수 (D-02 / D-03 / D-20 자동 차단).
     *
     * Gmail MCP ThreadMessage shape에 labelIds 필드가 없으므로(thread-loader.ts
     * L28) INBOX/SENT 라벨 대신 from 헤더 도메인 검사로 대체. 자사 도메인이
     * 아니면 inbound로 간주. from 헤더 누락 시 "inbound 아님"으로 처리하여
     * false-negative만 발생시킨다(팔로업이 1회 더 생기는 쪽이 상대방 답장을
     * 놓치는 것보다 안전).
     */

    import type { ThreadMessage } from './thread-loader.js';

    export const OWN_EMAIL_DOMAINS = ['@puzl.co.kr'] as const;

    export function isOwnMessage(message: Pick<ThreadMessage, 'from'>): boolean {
      const from = (message.from ?? '').toLowerCase();
      if (!from) return false;
      return OWN_EMAIL_DOMAINS.some((domain) => from.includes(domain));
    }

    /**
     * @param messages 시간순으로 정렬된 ThreadMessage 배열
     * @param sinceMs last outbound 시점 (엄격 > 비교)
     */
    export function hasInboundSince(
      messages: ThreadMessage[],
      sinceMs: number,
    ): boolean {
      return messages.some((m) => {
        const ts = parseInt(m.internalDate ?? '0', 10) || 0;
        if (ts <= sinceMs) return false;
        if (!m.from || m.from.trim() === '') return false;  // safe default
        if (isOwnMessage(m)) return false;
        return true;
      });
    }
    ```

    export interface ForbiddenPhraseCheckResult {
      safe: boolean;
      violations: Array<{ phrase: string; reason: string }>;
    }

    export function validateNoForbiddenPhrases(
      htmlBody: string,
    ): ForbiddenPhraseCheckResult {
      const normalized = (htmlBody ?? '').toLowerCase();
      const violations = FORBIDDEN_FOLLOWUP_PHRASES.filter(({ phrase }) =>

      );
      return { safe: violations.length === 0, violations };
