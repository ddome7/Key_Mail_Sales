```typescript
// UNSAFE_CHAR_MAP: Record<string, string>
// 위험 문자 -> 안전 대체 문자 매핑 (01-RESEARCH.md 기반):
const UNSAFE_CHAR_MAP: Record<string, string> = {
  '\u2514': '-',     // └ -> dash
  '\u2192': '',      // → -> remove (rewrite as text)
  '\u251C': '-',     // ├ -> dash
  '\u2500': '-',     // ─ -> dash
  '\u2502': '|',     // │ -> pipe
  '\u2510': '',      // ┐ -> remove
  '\u2518': '',      // ┘ -> remove
  '\u250C': '',      // ┌ -> remove
  '\u2013': '-',     // – (en dash) -> regular dash
  '\u2014': '-',     // — (em dash) -> regular dash
  '\u2018': "'",     // ' -> apostrophe
  '\u2019': "'",     // ' -> apostrophe
  '\u201C': '"',     // " -> quote
  '\u201D': '"',     // " -> quote
};

// SAFE_CHARS: 안전 확인된 문자 목록 (치환하지 않음)
const SAFE_CHARS = ['■', '●', '◎', '※', '★', '①', '②', '③', '✅'];

// sanitizeForGmail(html: string): string
// - UNSAFE_CHAR_MAP의 모든 키를 값으로 replaceAll
// - SAFE_CHARS는 건드리지 않음
// - 입력과 출력이 다른 경우 변경 내역 로그 (console.log)

// getSanitizationReport(original: string, sanitized: string): SanitizationReport
// - 어떤 문자가 몇 번 치환되었는지 리포트
// - { totalReplacements: number, details: [{char: string, count: number, replacement: string}] }
```
